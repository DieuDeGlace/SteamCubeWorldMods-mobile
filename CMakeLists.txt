cmake_minimum_required(VERSION 3.20)
project(SteamCubeWorldMods LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MSVC)
  message(FATAL_ERROR "Use clang-cl in an x64 Visual Studio tools environment")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
function(configure_mod target)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/cwmods")
  target_link_libraries(${target} PRIVATE d3d11 dxgi d3dcompiler user32 winmm)
endfunction()
add_library(WorldMotion SHARED WorldMotion/WorldMotion.cpp)
configure_mod(WorldMotion)
add_library(TerrainVariance SHARED TerrainVariance/TerrainVariance.cpp)
configure_mod(TerrainVariance)
add_library(FieldBosses SHARED FieldBosses/FieldBosses.cpp)
configure_mod(FieldBosses)
add_library(DualHair SHARED DualHair/DualHair.cpp)
configure_mod(DualHair)
add_library(WaterLilies SHARED WaterLilies/WaterLilies.cpp)
configure_mod(WaterLilies)
include(CTest)
option(ENABLE_GPU_TESTS "Enable tests requiring D3D11 support" OFF)
if(BUILD_TESTING)
  add_executable(WaterLiliesRulesTest WaterLilies/RulesTest.cpp)
  target_compile_options(WaterLiliesRulesTest PRIVATE /UNDEBUG)
  add_test(NAME WaterLiliesRules COMMAND WaterLiliesRulesTest)
  add_executable(WindRulesTest WorldMotion/WindRulesTest.cpp)
  configure_mod(WindRulesTest)
  target_compile_options(WindRulesTest PRIVATE /UNDEBUG)
  add_test(NAME WindRules COMMAND WindRulesTest)
  add_executable(TerrainRulesTest TerrainVariance/TerrainRulesTest.cpp)
  configure_mod(TerrainRulesTest)
  target_compile_options(TerrainRulesTest PRIVATE /UNDEBUG)
  add_test(NAME TerrainRules COMMAND TerrainRulesTest)
  add_executable(PlacementTest TerrainVariance/PlacementTest.cpp)
  configure_mod(PlacementTest)
  target_compile_options(PlacementTest PRIVATE /UNDEBUG)
  add_test(NAME Placement COMMAND PlacementTest)
  add_executable(FieldRulesTest FieldBosses/RulesTest.cpp)
  configure_mod(FieldRulesTest)
  target_compile_options(FieldRulesTest PRIVATE /UNDEBUG)
  add_test(NAME FieldRules COMMAND FieldRulesTest)
  add_executable(DualHairRulesTest DualHair/RulesTest.cpp)
  target_compile_options(DualHairRulesTest PRIVATE /UNDEBUG)
  add_test(NAME DualHairRules COMMAND DualHairRulesTest)
  if(ENABLE_GPU_TESTS)
  add_executable(GpuWindTest WorldMotion/GpuWindTest.cpp)
  configure_mod(GpuWindTest)
  target_compile_options(GpuWindTest PRIVATE /UNDEBUG)
  add_test(NAME GpuWind COMMAND GpuWindTest)
  endif()
endif()
